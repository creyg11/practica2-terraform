# 3. Verificar que el balanceador sí atiende en ambos entornos simultáneamente
curl -s -o /dev/null -w "PROD (8080) -> HTTP %{http_code}\n" http://127.0.0.1:8080/jugadores
curl -s -o /dev/null -w "DEV  (8081) -> HTTP %{http_code}\n" http://127.0.0.1:8081/jugadores

echo "--- 1. Estado normal (3 réplicas activas) ---"
for i in $(seq 1 6); do
  INST=$(curl -s http://127.0.0.1:8080/jugadores | grep -oE 'practica2-prod-web-[0-9]+' | head -n 1)
  echo "Petición $i -> $INST"
done

echo "--- 2. Caída de web-2 (el servicio mantiene HTTP 200 entre web-1 y web-3) ---"
docker stop practica2-prod-web-2
for i in $(seq 1 6); do
  CODE=$(curl -s -o /tmp/resp.html -w "%{http_code}" http://127.0.0.1:8080/jugadores)
  INST=$(grep -oE 'practica2-prod-web-[0-9]+' /tmp/resp.html | head -n 1)
  echo "Petición $i -> HTTP $CODE | Atendida por: $INST"
done

echo "--- 3. Recuperación de web-2 (reentra tras fail_timeout=5s) ---"
docker start practica2-prod-web-2
sleep 6
for i in $(seq 1 6); do
  INST=$(curl -s http://127.0.0.1:8080/jugadores | grep -oE 'practica2-prod-web-[0-9]+' | head -n 1)
  echo "Petición $i -> $INST"
done
